# --- 1. Rebuild ---
cd ~/Projects/LagoonSNESRecomp
cmake --build build-port --target LagoonSNESRecomp 2>&1 | tail -5

# --- 2. Set the options, then QUIT (240 needs a restart) ---
# Options > Comfort: Screen shake -> Off.  Pictures per second -> it shows "180 > 240".
# Quit the game. The next start runs at 240 with the shake off.
cd ~/Projects/LagoonSNESRecomp
./build-port/LagoonSNESRecomp --no-launcher --rom lagoon.sfc 2> settings.txt
grep "lagoon-settings\|lagoon-shake" settings.txt
# Expect: saved ScreenShake = 0, saved PicturesPerSecond = 240, screen shake OFF
